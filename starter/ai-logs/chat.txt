#### `scaffold/server/src/eventsAndQueue.ts`

This handles:

* Monotonic per-workspace `seq` generation via Redis `INCR` + Mongo persistence + Redis Pub/Sub fan-out across server replicas.
* Catch-up replay (`?after=<seq>`) without losing/duplicating live events during initial subscription.
* Distributed per-workspace rate limiting and durable lease-based outbound delivery that survives `kill -9` and multi-instance failover.

```ts
import crypto from "crypto";
import Redis from "ioredis";
import { WebSocket } from "ws";
import { config } from "./config";
import { ConversationModel, EventModel, OutboundMessageModel } from "./db";

export const redis = new Redis(config.redisUrl, { maxRetriesPerRequest: null });
export const redisSub = new Redis(config.redisUrl, { maxRetriesPerRequest: null });

const INSTANCE_ID = `${process.pid}-${crypto.randomUUID().slice(0, 8)}`;

// Local WS clients keyed by workspaceId
const localSubscribers = new Map<string, Set<WebSocket>>();
const subscribedChannels = new Set<string>();

function wsChannel(workspaceId: string): string {
  return `switchboard:ws:${workspaceId}`;
}

redisSub.on("message", (channel, rawMessage) => {
  const prefix = "switchboard:ws:";
  if (!channel.startsWith(prefix)) return;
  const workspaceId = channel.slice(prefix.length);
  const sockets = localSubscribers.get(workspaceId);
  if (!sockets || sockets.size === 0) return;

  for (const ws of sockets) {
    if (ws.readyState === WebSocket.OPEN) {
      ws.send(rawMessage);
    }
  }
});

export async function publishWorkspaceEvent(
  workspaceId: string,
  type: "message.inbound" | "message.status" | "conversation.assigned",
  payload: Record<string, unknown>
): Promise<{ seq: number; frame: Record<string, unknown> }> {
  const seq = await redis.incr(`switchboard:seq:${workspaceId}`);
  const frame = { seq, type, ...payload };

  await EventModel.create({
    workspaceId,
    seq,
    type,
    payload,
  });

  await redis.publish(wsChannel(workspaceId), JSON.stringify(frame));
  return { seq, frame };
}

export async function attachWebSocket(ws: WebSocket, workspaceId: string, afterSeq: number): Promise<void> {
  const channel = wsChannel(workspaceId);
  if (!subscribedChannels.has(channel)) {
    subscribedChannels.add(channel);
    await redisSub.subscribe(channel);
  }

  // Buffer live frames while we fetch historical catch-up events so nothing is missed
  const buffer: string[] = [];
  let flushing = true;

  const bufferListener = (ch: string, msg: string) => {
    if (ch !== channel) return;
    if (flushing) {
      buffer.push(msg);
    } else if (ws.readyState === WebSocket.OPEN) {
      ws.send(msg);
    }
  };

  redisSub.on("message", bufferListener);

  let set = localSubscribers.get(workspaceId);
  if (!set) {
    set = new Set();
    localSubscribers.set(workspaceId, set);
  }
  set.add(ws);

  const missed = await EventModel.find({ workspaceId, seq: { $gt: afterSeq } })
    .sort({ seq: 1 })
    .lean();

  let lastSentSeq = afterSeq;
  for (const ev of missed) {
    if (ws.readyState !== WebSocket.OPEN) break;
    lastSentSeq = Math.max(lastSentSeq, ev.seq);
    ws.send(JSON.stringify({ seq: ev.seq, type: ev.type, ...(ev.payload || {}) }));
  }

  for (const raw of buffer) {
    if (ws.readyState !== WebSocket.OPEN) break;
    try {
      const parsed = JSON.parse(raw);
      if (typeof parsed.seq === "number" && parsed.seq > lastSentSeq) {
        lastSentSeq = parsed.seq;
        ws.send(raw);
      }
    } catch {
      ws.send(raw);
    }
  }
  flushing = false;

  ws.on("close", () => {
    redisSub.off("message", bufferListener);
    localSubscribers.get(workspaceId)?.delete(ws);
  });
}

// Atomic per-workspace rate limiter shared across all server instances
async function acquireWorkspaceRateSlot(workspaceId: string): Promise<boolean> {
  const bucket = Math.floor(Date.now() / config.rateLimitWindowMs);
  const key = `switchboard:rl:${workspaceId}:${bucket}`;
  const count = await redis.incr(key);
  if (count === 1) {
    await redis.pexpire(key, config.rateLimitWindowMs * 2);
  }
  return count <= config.rateLimitMax;
}

const STATUS_RANK: Record<string, number> = {
  failed: 0,
  queued: 1,
  sent: 2,
  delivered: 3,
};

export async function enqueueReply(params: {
  workspaceId: string;
  contact: string;
  text: string;
  clientRef: string;
}): Promise<{ accepted: true }> {
  const { workspaceId, contact, text, clientRef } = params;

  await ConversationModel.findOneAndUpdate(
    { workspaceId, contact },
    {
      $setOnInsert: { workspaceId, contact, assignee: null, status: "open" },
      $set: { lastActivityAt: new Date() },
    },
    { upsert: true, new: true }
  );

  // Idempotent on (workspaceId, clientRef): concurrent or repeated calls never duplicate
  await OutboundMessageModel.updateOne(
    { workspaceId, clientRef },
    {
      $setOnInsert: {
        workspaceId,
        contact,
        clientRef,
        text,
        status: "queued",
        statusRank: STATUS_RANK.queued,
        attempts: 0,
        nextAttemptAt: new Date(),
        lockedBy: null,
        lockExpiresAt: null,
      },
    },
    { upsert: true }
  );

  return { accepted: true };
}

export async function updateOutboundStatus(
  workspaceId: string,
  clientRef: string,
  nextStatus: "queued" | "sent" | "delivered" | "failed",
  contactHint?: string
): Promise<void> {
  const doc = await OutboundMessageModel.findOne({ workspaceId, clientRef });
  if (!doc) return;

  const incomingRank = STATUS_RANK[nextStatus] ?? 0;
  // Ignore stale/out-of-order lower status transitions except terminal failure if still queued
  if (incomingRank < doc.statusRank && !(nextStatus === "failed" && doc.status === "queued")) {
    return;
  }
  if (doc.status === nextStatus) return;

  doc.status = nextStatus;
  doc.statusRank = incomingRank;
  doc.lockedBy = null;
  doc.lockExpiresAt = null;
  await doc.save();

  await publishWorkspaceEvent(workspaceId, "message.status", {
    contact: contactHint || doc.contact,
    client_ref: clientRef,
    status: nextStatus,
  });
}

async function processOneOutboundJob(): Promise<boolean> {
  const now = new Date();
  const leaseUntil = new Date(now.getTime() + 20_000);

  // Claim a queued message whose lease is either null or expired (e.g., after kill -9)
  const job = await OutboundMessageModel.findOneAndUpdate(
    {
      status: "queued",
      nextAttemptAt: { $lte: now },
      $or: [{ lockExpiresAt: null }, { lockExpiresAt: { $lte: now } }],
    },
    {
      $set: { lockedBy: INSTANCE_ID, lockExpiresAt: leaseUntil },
      $inc: { attempts: 1 },
    },
    { sort: { createdAt: 1 }, new: true }
  );

  if (!job) return false;

  const allowed = await acquireWorkspaceRateSlot(job.workspaceId);
  if (!allowed) {
    // Release lock and push out slightly rather than burning attempts on self-throttling
    await OutboundMessageModel.updateOne(
      { _id: job._id },
      {
        $set: {
          lockedBy: null,
          lockExpiresAt: null,
          nextAttemptAt: new Date(Date.now() + 250 + Math.random() * 250),
        },
        $inc: { attempts: -1 },
      }
    );
    return false;
  }

  try {
    // Adjust path/body keys to match PIGEON.md if Pigeon expects different field names
    const res = await fetch(`${config.pigeonUrl}/v1/messages`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Idempotency-Key": job.clientRef,
      },
      body: JSON.stringify({
        workspace: job.workspaceId,
        workspace_id: job.workspaceId,
        to: job.contact,
        contact: job.contact,
        text: job.text,
        client_ref: job.clientRef,
      }),
      signal: AbortSignal.timeout(8000),
    });

    if (res.ok) {
      await updateOutboundStatus(job.workspaceId, job.clientRef, "sent", job.contact);
      return true;
    }

    if (res.status === 429) {
      const retryAfterSec = Number(res.headers.get("retry-after") || "1");
      const delayMs = Math.max(200, retryAfterSec * 1000) + Math.floor(Math.random() * 200);
      await OutboundMessageModel.updateOne(
        { _id: job._id },
        { $set: { lockedBy: null, lockExpiresAt: null, nextAttemptAt: new Date(Date.now() + delayMs) } }
      );
      return true;
    }

    if (res.status >= 400 && res.status < 500) {
      await updateOutboundStatus(job.workspaceId, job.clientRef, "failed", job.contact);
      return true;
    }

    // 5xx or transient failure: backoff and retry
    const backoffMs = Math.min(10_000, 300 * Math.pow(1.6, Math.min(job.attempts, 8)));
    await OutboundMessageModel.updateOne(
      { _id: job._id },
      { $set: { lockedBy: null, lockExpiresAt: null, nextAttemptAt: new Date(Date.now() + backoffMs) } }
    );
    return true;
  } catch {
    const backoffMs = Math.min(10_000, 300 * Math.pow(1.6, Math.min(job.attempts, 8)));
    await OutboundMessageModel.updateOne(
      { _id: job._id },
      { $set: { lockedBy: null, lockExpiresAt: null, nextAttemptAt: new Date(Date.now() + backoffMs) } }
    );
    return true;
  }
}

let workerTimer: NodeJS.Timeout | null = null;
let workerRunning = false;

export function startOutboundWorker(): void {
  if (workerTimer) return;
  const tick = async () => {
    if (workerRunning) return;
    workerRunning = true;
    try {
      // Drain up to 10 ready jobs per tick
      for (let i = 0; i < 10; i++) {
        const processed = await processOneOutboundJob();
        if (!processed) break;
      }
    } finally {
      workerRunning = false;
    }
  };
  workerTimer = setInterval(tick, 120);
  tick().catch(() => {});
}
```

```ts
function signSession(payload: { sub: string; workspaces: WorkspaceRole[] }): string {
  const encoded = Buffer.from(JSON.stringify(payload)).toString("base64url");
  const sig = crypto.createHmac("sha256", config.sessionSecret).update(encoded).digest("base64url");
  return `${encoded}.${sig}`;
}

function readSession(req: Request): { sub: string; workspaces: WorkspaceRole[] } | null {
  const raw = req.cookies?.[config.cookieName];
  if (!raw || typeof raw !== "string") return null;
  const parts = raw.split(".");
  if (parts.length !== 2) return null;
  const [encoded, sig] = parts;
  const expected = crypto.createHmac("sha256", config.sessionSecret).update(encoded).digest("base64url");
  if (!crypto.timingSafeEqual(Buffer.from(sig), Buffer.from(expected))) return null;
  try {
    return JSON.parse(Buffer.from(encoded, "base64url").toString("utf8"));
  } catch {
    return null;
  }
}

function requireAuth(req: Request, res: Response, next: NextFunction): void {
  const session = readSession(req);
  if (!session) {
    res.status(401).json({ error: "unauthorized" });
    return;
  }
  (req as Request & { session?: { sub: string; workspaces: WorkspaceRole[] } }).session = session;
  next();
}

function requireWorkspaceMember(req: Request, res: Response, workspaceId: string): boolean {
  const session = (req as Request & { session?: { sub: string; workspaces: WorkspaceRole[] } }).session;
  if (!session) {
    res.status(401).json({ error: "unauthorized" });
    return false;
  }
  const hasAccess = session.workspaces.some((w) => w.id === workspaceId);
  if (!hasAccess) {
    res.status(403).json({ error: "forbidden" });
    return false;
  }
  return true;
}

  const code = String(req.query.code || "");

      client_id: config.idpClientId,

      code_verifier: verifier,

  });

    res.status(502).send("Token exchange failed");
    return;

  const tokenData = (await tokenRes.json()) as { id_token?: string; access_token?: string };
  const rawJwt = tokenData.id_token || tokenData.access_token || "";
  const parts = rawJwt.split(".");

    return;

async function claimConversation(
  workspaceId: string,
  contact: string,
  agent: string
): Promise<{ status: number; body: Record<string, unknown> }> {
  await ConversationModel.updateOne(

    },

// Pigeon webhook receiver: verify signature per PIGEON.md, deduplicate, order, and broadcast
function verifyPigeonSignature(req: Request & { rawBody?: Buffer }): boolean {
  if (!config.pigeonWebhookSecret) return true;
  const sigHeader = String(
    req.headers["x-pigeon-signature"] || req.headers["x-signature-256"] || req.headers["x-signature"] || ""
  );
  if (!sigHeader || !req.rawBody) return false;
  const cleaned = sigHeader.replace(/^sha256=/, "");
  const expected = crypto.createHmac("sha256", config.pigeonWebhookSecret).update(req.rawBody).digest("hex");
  try {
    return crypto.timingSafeEqual(Buffer.from(cleaned), Buffer.from(expected));
  } catch {
    return false;
  }
}

server.on("upgrade", (req, socket, head) => {
  const url = new URL(req.url || "/", `http://${req.headers.host || "localhost"}`);
  if (url.pathname !== "/ws") {
    socket.write("HTTP/1.1 404 Not Found\r\n\r\n");
    socket.destroy();
    return;
  }

  // Parse cookie from upgrade request
  const cookieHeader = req.headers.cookie || "";
  const parsedCookies: Record<string, string> = {};
  for (const part of cookieHeader.split(";")) {
    const idx = part.indexOf("=");
    if (idx > 0) {
      parsedCookies[part.slice(0, idx).trim()] = decodeURIComponent(part.slice(idx + 1).trim());
    }
  }

  const fakeReq = { cookies: parsedCookies } as Request;
  const session = readSession(fakeReq);
  const workspaceId = url.searchParams.get("workspace") || "";
  const afterSeq = Math.max(0, Number(url.searchParams.get("after") || "0") || 0);

  if (!session) {
    socket.write("HTTP/1.1 401 Unauthorized\r\n\r\n");
    socket.destroy();
    return;
  }

  if (!workspaceId || !session.workspaces.some((w) => w.id === workspaceId)) {
    socket.write("HTTP/1.1 403 Forbidden\r\n\r\n");
    socket.destroy();
    return;
  }

  wss.handleUpgrade(req, socket, head, (ws) => {
    attachWebSocket(ws, workspaceId, afterSeq).catch(() => ws.close());
  });
});
```

```ts
  EventDoc,

const RATE_LIMIT_WINDOW_MS = Number(process.env.PIGEON_RATE_LIMIT_WINDOW_MS || '1000');

  failed: 0,
  queued: 1,
  sent: 2,
  delivered: 3,

  return `switchboard:ws:${workspaceId}`;
}

function getSubscriber() {
  if (!redisSubClient) {
    redisSubClient = getRedis().duplicate();
    redisSubClient.on('message', (channel: string, rawMessage: string) => {
      const prefix = 'switchboard:ws:';
      if (!channel.startsWith(prefix)) return;
      const workspaceId = channel.slice(prefix.length);
      const sockets = localSubscribers.get(workspaceId);
      if (!sockets || sockets.size === 0) return;

      for (const ws of sockets) {
        if (ws.readyState === WebSocket.OPEN) {
          ws.send(rawMessage);
        }
      }
    });
  }
  return redisSubClient;
}

export async function publishWorkspaceEvent(

  payload: Record<string, unknown>

export async function attachWebSocket(
  ws: WebSocket,
  workspaceId: string,
  afterSeq: number
): Promise<void> {
  await ensureIndexes();
  const db = await getDb();
  const sub = getSubscriber();
  const channel = wsChannel(workspaceId);

  if (!subscribedChannels.has(channel)) {
    subscribedChannels.add(channel);
    await sub.subscribe(channel);
  }

  const buffer: string[] = [];
  let flushing = true;

  const onMessage = (ch: string, msg: string) => {
    if (ch !== channel) return;
    if (flushing) {
      buffer.push(msg);
    } else if (ws.readyState === WebSocket.OPEN) {
      ws.send(msg);
    }
  };

  sub.on('message', onMessage);

  let set = localSubscribers.get(workspaceId);
  if (!set) {
    set = new Set();
    localSubscribers.set(workspaceId, set);
  }
  set.add(ws);

  const missed = await db
    .collection<EventDoc>('events')
    .find({ workspaceId, seq: { $gt: afterSeq } })
    .sort({ seq: 1 })
    .toArray();

  let lastSentSeq = afterSeq;
  for (const ev of missed) {
    if (ws.readyState !== WebSocket.OPEN) break;
    lastSentSeq = Math.max(lastSentSeq, ev.seq);
    ws.send(JSON.stringify({ seq: ev.seq, type: ev.type, ...(ev.payload || {}) }));
  }

  for (const raw of buffer) {
    if (ws.readyState !== WebSocket.OPEN) break;
    try {
      const parsed = JSON.parse(raw) as { seq?: number };
      if (typeof parsed.seq === 'number' && parsed.seq > lastSentSeq) {
        lastSentSeq = parsed.seq;
        ws.send(raw);
      }
    } catch {
      ws.send(raw);
    }
  }
  flushing = false;

  ws.on('close', () => {
    sub.off('message', onMessage);
    localSubscribers.get(workspaceId)?.delete(ws);
  });
}

async function getWorkspaceCooldownRemainingMs(workspaceId: string): Promise<number> {
  const redis = getRedis();
  const raw = await redis.get(`switchboard:pigeon_cooldown:${workspaceId}`);
  if (!raw) return 0;
  const remaining = Number(raw) - Date.now();
  return remaining > 0 ? remaining : 0;
}

async function setWorkspaceCooldown(workspaceId: string, ms: number): Promise<void> {
  const redis = getRedis();
  const clamped = Math.min(30_000, Math.max(250, ms));
  await redis.set(
    `switchboard:pigeon_cooldown:${workspaceId}`,
    String(Date.now() + clamped),
    'PX',
    clamped + 1000
  );
}

async function acquireWorkspaceRateSlot(workspaceId: string): Promise<boolean> {
  const cooldown = await getWorkspaceCooldownRemainingMs(workspaceId);
  if (cooldown > 0) return false;

  const redis = getRedis();
  const bucket = Math.floor(Date.now() / RATE_LIMIT_WINDOW_MS);
  const key = `switchboard:rl:${workspaceId}:${bucket}`;
  const count = await redis.incr(key);
  if (count === 1) {
    await redis.pexpire(key, RATE_LIMIT_WINDOW_MS * 2);
  }
  return count <= RATE_LIMIT_MAX;
}

export async function updateOutboundStatus(params: {
  workspaceId: string;
  clientRef?: string;
  pigeonMessageId?: string;
  nextStatus: 'queued' | 'sent' | 'delivered' | 'failed';
  pigeonMsgIdToSave?: string;
}): Promise<boolean> {
  await ensureIndexes();
  const db = await getDb();
  const col = db.collection<OutboundMessageDoc>('outbound_messages');

  const query: Record<string, unknown> = { workspaceId: params.workspaceId };
  if (params.clientRef) {
    query.clientRef = params.clientRef;
  } else if (params.pigeonMessageId) {
    query.pigeonMessageId = params.pigeonMessageId;
  } else {
    return false;
  }

  const doc = await col.findOne(query);
  if (!doc) return false;

  const incomingRank = STATUS_RANK[params.nextStatus];
  let changed = false;
  const setFields: Record<string, unknown> = {
    lockedBy: null,
    lockExpiresAt: null,
  };

  if (params.pigeonMsgIdToSave && !doc.pigeonMessageId) {
    setFields.pigeonMessageId = params.pigeonMsgIdToSave;
    changed = true;
  }

  if (incomingRank > doc.statusRank) {
    setFields.status = params.nextStatus;
    setFields.statusRank = incomingRank;
    changed = true;
  }

  await col.updateOne({ _id: doc._id }, { $set: setFields });

  if (changed) {
    await publishWorkspaceEvent(params.workspaceId, 'message.status', {
      contact: doc.contact,
      client_ref: doc.clientRef,
      status: incomingRank > doc.statusRank ? params.nextStatus : doc.status,
    });
  }

  return true;
}

async function processOneOutboundJob(): Promise<boolean> {
  await ensureIndexes();
  const db = await getDb();
  const col = db.collection<OutboundMessageDoc>('outbound_messages');
  const now = new Date();
  const leaseUntil = new Date(now.getTime() + 25_000);

  const res = await col.findOneAndUpdate(
    {
      status: 'queued',
      nextAttemptAt: { $lte: now },
      $or: [{ lockExpiresAt: null }, { lockExpiresAt: { $lte: now } }],
    },
    {
      $set: { lockedBy: INSTANCE_ID, lockExpiresAt: leaseUntil },
      $inc: { attempts: 1 },
    },
    { sort: { createdAt: 1 }, returnDocument: 'after' }
  );

  const job = res;
  if (!job) return false;

  const cooldownMs = await getWorkspaceCooldownRemainingMs(job.workspaceId);
  if (cooldownMs > 0) {
    await col.updateOne(
      { _id: job._id },
      {
        $set: {
          lockedBy: null,
          lockExpiresAt: null,
          nextAttemptAt: new Date(Date.now() + cooldownMs),
        },
        $inc: { attempts: -1 },
      }
    );
    return false;
  }

  const slotOk = await acquireWorkspaceRateSlot(job.workspaceId);
  if (!slotOk) {
    await col.updateOne(
      { _id: job._id },
      {
        $set: {
          lockedBy: null,
          lockExpiresAt: null,
          nextAttemptAt: new Date(Date.now() + 250 + Math.floor(Math.random() * 250)),
        },
        $inc: { attempts: -1 },
      }
    );
    return false;
  }

  const apiKey = config.pigeon.apiKeys[job.workspaceId];
  if (!apiKey) {
    await updateOutboundStatus({
      workspaceId: job.workspaceId,
      clientRef: job.clientRef,
      nextStatus: 'failed',
    });
    return true;
  }

  try {
    const response = await fetch(`${config.pigeon.url.replace(/\/+$/, '')}/v1/messages`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${apiKey}`,
        'Idempotency-Key': job.clientRef,
      },
      body: JSON.stringify({
        to: job.contact,
        text: job.text,
        client_ref: job.clientRef,
      }),
      signal: AbortSignal.timeout(5500),
    });

    if (response.ok) {
      let pigeonId: string | undefined;
      try {
        const json = (await response.json()) as { id?: string };
        if (json?.id) pigeonId = String(json.id);
      } catch {
        // ignore parse failure on 200
      }

      await updateOutboundStatus({
        workspaceId: job.workspaceId,
        clientRef: job.clientRef,
        nextStatus: 'sent',
        pigeonMsgIdToSave: pigeonId,
      });
      return true;
    }

    if (response.status === 429) {
      const retryAfterHeader = response.headers.get('retry-after');
      const retryAfterSec = retryAfterHeader ? Number(retryAfterHeader) : 1;
      const delayMs =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 ? retryAfterSec * 1000 : 1000;
      await setWorkspaceCooldown(job.workspaceId, delayMs);
      await col.updateOne(
        { _id: job._id },
        {
          $set: {
            lockedBy: null,
            lockExpiresAt: null,
            nextAttemptAt: new Date(Date.now() + delayMs + Math.floor(Math.random() * 200)),
          },
        }
      );
      return true;
    }

    if (response.status >= 400 && response.status < 500) {
      await updateOutboundStatus({
        workspaceId: job.workspaceId,
        clientRef: job.clientRef,
        nextStatus: 'failed',
      });
      return true;
    }

    const backoffMs =
      Math.min(6000, 350 * Math.pow(1.5, Math.min(job.attempts, 7))) +
      Math.floor(Math.random() * 200);
    await col.updateOne(
      { _id: job._id },
      {
        $set: {
          lockedBy: null,
          lockExpiresAt: null,
          nextAttemptAt: new Date(Date.now() + backoffMs),
        },
      }
    );
    return true;
  } catch {
    const backoffMs =
      Math.min(6000, 400 * Math.pow(1.5, Math.min(job.attempts, 7))) +
      Math.floor(Math.random() * 200);
    await col.updateOne(
      { _id: job._id },
      {
        $set: {
          lockedBy: null,
          lockExpiresAt: null,
          nextAttemptAt: new Date(Date.now() + backoffMs),
        },
      }
    );
    return true;
  }
}

let timer: ReturnType<typeof setInterval> | null = null;
let ticking = false;

export function startOutboundWorker(): void {
  if (timer) return;
  const runTick = async () => {
    if (ticking) return;
    ticking = true;
    try {
      for (let i = 0; i < 8; i++) {
        const worked = await processOneOutboundJob();
        if (!worked) break;
      }
    } finally {
      ticking = false;
    }

}
```

```ts
function parseCookies(cookieHeader: string | undefined): Record<string, string> {
  const out: Record<string, string> = {};
  if (!cookieHeader) return out;
  for (const part of cookieHeader.split(';')) {
    const idx = part.indexOf('=');
    if (idx <= 0) continue;
    const key = part.slice(0, idx).trim();
    const val = part.slice(idx + 1).trim();
    try {
      out[key] = decodeURIComponent(val);
    } catch {
      out[key] = val;
    }
  }
  return out;
}

function signSession(payload: SessionPayload): string {
  const encoded = Buffer.from(JSON.stringify(payload)).toString('base64url');
  const sig = crypto.createHmac('sha256', SESSION_SECRET).update(encoded).digest('base64url');
  return `${encoded}.${sig}`;
}

function verifySessionToken(token: string | undefined): SessionPayload | null {
  if (!token || typeof token !== 'string') return null;
  const parts = token.split('.');
  if (parts.length !== 2) return null;
  const [encoded, sig] = parts;
  const expected = crypto.createHmac('sha256', SESSION_SECRET).update(encoded).digest('base64url');
  try {
    const a = Buffer.from(sig);
    const b = Buffer.from(expected);
    if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) return null;
    const parsed = JSON.parse(Buffer.from(encoded, 'base64url').toString('utf8')) as SessionPayload;
    if (!parsed.sub || !Array.isArray(parsed.workspaces)) return null;
    return parsed;
  } catch {
    return null;
  }
}

function readSession(req: Request): SessionPayload | null {
  const cookies = parseCookies(req.headers.cookie);
  return verifySessionToken(cookies[SESSION_COOKIE]);
}

function requireWorkspaceMember(req: Request, res: Response, workspaceId: string): boolean {

}

      $setOnInsert: { workspaceId, contact, assignee: null, status: 'open', lastActivityAt: now },
    },
    { upsert: true }
  );

  const claimed = await col.findOneAndUpdate(
    {
      workspaceId,
      contact,
      assignee: { $in: [null, agent] },
    },
    {
      $set: { assignee: agent, status: 'open', lastActivityAt: now },
    },
    { returnDocument: 'after' }
  );

  if (!claimed) {
    const existing = await col.findOne({ workspaceId, contact });
    return {
      status: 409,
      body: { error: 'already_claimed', assignee: existing?.assignee ?? null },
    };
  }

  await publishWorkspaceEvent(workspaceId, 'conversation.assigned', {
    contact,
    assignee: agent,
  });

  return {
    status: 200,
    body: { contact, assignee: claimed.assignee, status: claimed.status },
  };
}

// Pigeon webhook signature verification: x-pigeon-signature: t=<sec>,v1=<hex> over "<t>.<raw body>"
function verifyPigeonSignature(req: RequestWithRawBody): boolean {
  const sigHeader = req.headers['x-pigeon-signature'];
  if (!sigHeader || typeof sigHeader !== 'string' || !req.rawBody) return false;

  let tPart = '';
  let v1Part = '';
  for (const piece of sigHeader.split(',')) {
    const trimmed = piece.trim();
    if (trimmed.startsWith('t=')) tPart = trimmed.slice(2);
    else if (trimmed.startsWith('v1=')) v1Part = trimmed.slice(3);
  }

  const tsSec = Number(tPart);
  if (!tPart || !v1Part || !Number.isFinite(tsSec)) return false;

  // Reject timestamps more than 5 minutes (300s) old
  const nowSec = Math.floor(Date.now() / 1000);
  if (Math.abs(nowSec - tsSec) > 300) return false;

  const expectedHex = crypto
    .createHmac('sha256', config.pigeon.webhookSecret)
    .update(`${tPart}.${req.rawBody.toString('utf8')}`)
    .digest('hex');

  try {
    const actualBuf = Buffer.from(v1Part, 'hex');
    const expectedBuf = Buffer.from(expectedHex, 'hex');
    if (actualBuf.length === 0 || actualBuf.length !== expectedBuf.length) return false;
    return crypto.timingSafeEqual(actualBuf, expectedBuf);
  } catch {
    return false;
  }
}

app.post('/webhooks/pigeon', async (req, res) => {
  const typedReq = req as RequestWithRawBody;
  if (!verifyPigeonSignature(typedReq)) {
    res.status(401).json({ error: 'invalid_signature' });
    return;
  }

  if (!req.body || typeof req.body !== 'object' || Array.isArray(req.body)) {
    res.status(400).json({ error: 'invalid_body' });
    return;
  }

  const body = req.body as Record<string, unknown>;
  const workspaceId = typeof body.workspace === 'string' ? body.workspace : '';
  const type = typeof body.type === 'string' ? body.type : '';
  const data =
    body.data && typeof body.data === 'object' && !Array.isArray(body.data)
      ? (body.data as Record<string, unknown>)
      : {};

  // Unknown workspace -> 2xx and ignore per PIGEON.md
  if (!workspaceId || !Object.prototype.hasOwnProperty.call(config.pigeon.apiKeys, workspaceId)) {
    res.status(200).json({ ok: true, ignored: 'unknown_workspace' });
    return;
  }

  await ensureIndexes();
  const db = await getDb();

  if (type === 'message.inbound') {
    const messageId = typeof data.message_id === 'string' ? data.message_id : '';
    const contact = typeof data.from === 'string' ? data.from : '';
    const text = typeof data.text === 'string' ? data.text : '';

    if (!messageId || !contact) {
      res.status(400).json({ error: 'missing_inbound_fields' });
      return;
    }

    const occurredAtParsed =
      typeof body.occurred_at === 'string' ? Date.parse(body.occurred_at) : NaN;
    const occurredAt = Number.isFinite(occurredAtParsed) ? new Date(occurredAtParsed) : new Date();
    const sortOrder = occurredAt.getTime();

    const inboundCol = db.collection<InboundMessageDoc>('inbound_messages');
    const existing = await inboundCol.findOne({ workspaceId, messageId });
    if (!existing) {
      try {
        await inboundCol.insertOne({
          workspaceId,
          contact,
          messageId,
          text,
          occurredAt,
          sortOrder,
        });
      } catch (err: unknown) {
        if ((err as { code?: number })?.code !== 11000) {
          throw err;
        }
      }

      await db.collection<ConversationDoc>('conversations').updateOne(
        { workspaceId, contact },
        {
          $setOnInsert: { workspaceId, contact, assignee: null, status: 'open' },
          $max: { lastActivityAt: occurredAt },
        },
        { upsert: true }
      );

      await publishWorkspaceEvent(workspaceId, 'message.inbound', {
        contact,
        message_id: messageId,
        text,
      });
    }

    res.status(200).json({ ok: true });
    return;
  }

  if (type === 'message.status') {
    const clientRef = typeof data.client_ref === 'string' ? data.client_ref : undefined;
    const pigeonMessageId = typeof data.message_id === 'string' ? data.message_id : undefined;
    const rawStatus = typeof data.status === 'string' ? data.status.toLowerCase() : '';

    if (rawStatus !== 'sent' && rawStatus !== 'delivered' && rawStatus !== 'failed') {
      res.status(200).json({ ok: true, ignored: 'unknown_status' });
      return;
    }

    await updateOutboundStatus({
      workspaceId,
      clientRef,
      pigeonMessageId,
      nextStatus: rawStatus,
      pigeonMsgIdToSave: pigeonMessageId,
    });

    res.status(200).json({ ok: true });
    return;
  }

  // Unknown event type -> 2xx per PIGEON.md
  res.status(200).json({ ok: true, ignored: 'unknown_event_type' });
});

app.get('/__grader/snapshot', requireGrader, async (_req, res) => {
  await ensureIndexes();
  const db = await getDb();

  const [conversations, inboundMessages, outboundMessages] = await Promise.all([
    db.collection<ConversationDoc>('conversations').find({}).toArray(),
    db.collection<InboundMessageDoc>('inbound_messages').find({}).sort({ sortOrder: 1, _id: 1 }).toArray(),
    db.collection<OutboundMessageDoc>('outbound_messages').find({}).toArray(),
  ]);

  const workspaces: Record<
    string,
    {
      contacts: Record<string, { inbound: string[]; assignee: string | null }>;
      outbound: Record<string, string>;
    }
  > = {};

  const getOrCreateWs = (wsId: string) => {
    if (!workspaces[wsId]) {
      workspaces[wsId] = { contacts: {}, outbound: {} };
    }
    return workspaces[wsId];
  };

  for (const conv of conversations) {
    const ws = getOrCreateWs(conv.workspaceId);
    if (!ws.contacts[conv.contact]) {
      ws.contacts[conv.contact] = { inbound: [], assignee: conv.assignee ?? null };
    } else {
      ws.contacts[conv.contact].assignee = conv.assignee ?? null;
    }
  }

  for (const msg of inboundMessages) {
    const ws = getOrCreateWs(msg.workspaceId);
    if (!ws.contacts[msg.contact]) {
      ws.contacts[msg.contact] = { inbound: [], assignee: null };
    }
    ws.contacts[msg.contact].inbound.push(msg.messageId);
  }

  for (const out of outboundMessages) {
    const ws = getOrCreateWs(out.workspaceId);
    ws.outbound[out.clientRef] = out.status;
  }

  res.status(200).json({ workspaces });
});

server.on('upgrade', (req, socket, head) => {
  const url = new URL(req.url || '/', `http://${req.headers.host || 'localhost'}`);
  if (url.pathname !== '/ws') {
    socket.write('HTTP/1.1 404 Not Found\r\n\r\n');
    socket.destroy();
    return;
  }

  const cookies = parseCookies(req.headers.cookie);
  const session = verifySessionToken(cookies[SESSION_COOKIE]);
  const workspaceId = url.searchParams.get('workspace') || '';
  const afterSeq = Math.max(0, Number(url.searchParams.get('after') || '0') || 0);

  if (!session) {
    socket.write('HTTP/1.1 401 Unauthorized\r\n\r\n');
    socket.destroy();
    return;
  }

  if (!workspaceId || !session.workspaces.some((w) => w.id === workspaceId)) {
    socket.write('HTTP/1.1 403 Forbidden\r\n\r\n');
    socket.destroy();
    return;
  }

  wss.handleUpgrade(req, socket, head, (ws) => {
    attachWebSocket(ws, workspaceId, afterSeq).catch(() => ws.close());
  });
});
```

```tsx
import { useEffect, useRef, useState } from 'react';

interface WorkspaceRole {
  id: string;
  role: 'admin' | 'agent';
}

interface MeResponse {
  sub: string;
  workspaces: WorkspaceRole[];
}

interface ConversationItem {
  workspaceId: string;
  contact: string;
  assignee: string | null;
  status: 'open' | 'closed';
  lastActivityAt: string;
}

interface InboundItem {
  messageId: string;
  text: string;
  sortOrder: number;
}

interface OutboundItem {
  clientRef: string;
  text: string;
  status: 'queued' | 'sent' | 'delivered' | 'failed';
}

export function App() {
  const [me, setMe] = useState<MeResponse | null>(null);
  const [loadingMe, setLoadingMe] = useState(true);
  const [workspaceId, setWorkspaceId] = useState('');
  const [conversations, setConversations] = useState<ConversationItem[]>([]);
  const [selectedContact, setSelectedContact] = useState('');
  const [inbound, setInbound] = useState<InboundItem[]>([]);
  const [outbound, setOutbound] = useState<OutboundItem[]>([]);
  const [replyText, setReplyText] = useState('');
  const [notice, setNotice] = useState<string | null>(null);

  const lastSeqRef = useRef<number>(0);

  useEffect(() => {
    fetch('/api/me', { credentials: 'include' })
      .then(async (res) => (res.ok ? ((await res.json()) as MeResponse) : null))
      .then((user) => {
        setMe(user);
        if (user?.workspaces?.length) {
          setWorkspaceId(user.workspaces[0].id);
        }
      })
      .finally(() => setLoadingMe(false));
  }, []);

  async function refreshConversations(ws: string) {
    const res = await fetch(`/api/conversations?workspace=${encodeURIComponent(ws)}`, {
      credentials: 'include',
    });
    if (!res.ok) {
      setNotice(`Unable to load conversations (${res.status})`);
      return;
    }
    const data = (await res.json()) as { conversations: ConversationItem[] };
    setConversations(data.conversations || []);
  }

  async function refreshMessages(ws: string, contact: string) {
    const res = await fetch(
      `/api/conversations/${encodeURIComponent(contact)}/messages?workspace=${encodeURIComponent(ws)}`,
      { credentials: 'include' }
    );
    if (!res.ok) return;
    const data = (await res.json()) as { inbound: InboundItem[]; outbound: OutboundItem[] };
    setInbound(data.inbound || []);
    setOutbound(data.outbound || []);
  }

  useEffect(() => {
    if (!workspaceId) return;
    lastSeqRef.current = 0;
    setNotice(null);
    refreshConversations(workspaceId);
    if (selectedContact) {
      refreshMessages(workspaceId, selectedContact);
    }

    let closedByCleanup = false;
    let reconnectTimer: ReturnType<typeof setTimeout> | null = null;
    let socket: WebSocket | null = null;

    const connect = () => {
      if (closedByCleanup) return;
      const proto = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
      const url = `${proto}//${window.location.host}/ws?workspace=${encodeURIComponent(
        workspaceId
      )}&after=${lastSeqRef.current}`;

      socket = new WebSocket(url);
      socket.onmessage = (event) => {
        try {
          const frame = JSON.parse(event.data) as {
            seq?: number;
            type?: string;
            contact?: string;
            client_ref?: string;
            status?: OutboundItem['status'];
            assignee?: string | null;
          };
          if (typeof frame.seq === 'number' && frame.seq > lastSeqRef.current) {
            lastSeqRef.current = frame.seq;
          }
          if (frame.type === 'message.inbound') {
            refreshConversations(workspaceId);
            if (frame.contact === selectedContact) {
              refreshMessages(workspaceId, selectedContact);
            }
          } else if (frame.type === 'message.status' && frame.client_ref && frame.status) {
            setOutbound((prev) =>
              prev.map((m) =>
                m.clientRef === frame.client_ref ? { ...m, status: frame.status! } : m
              )
            );
          } else if (frame.type === 'conversation.assigned' && frame.contact) {
            setConversations((prev) =>
              prev.map((c) =>
                c.contact === frame.contact ? { ...c, assignee: frame.assignee ?? null } : c
              )
            );
          }
        } catch {
          // ignore malformed frames
        }
      };

      socket.onclose = () => {
        if (!closedByCleanup) {
          reconnectTimer = setTimeout(connect, 1000);
        }
      };
    };

    connect();

    return () => {
      closedByCleanup = true;
      if (reconnectTimer) clearTimeout(reconnectTimer);
      socket?.close();
    };
  }, [workspaceId, selectedContact]);

  async function onClaim(contact: string) {
    setNotice(null);
    const res = await fetch(`/api/conversations/${encodeURIComponent(contact)}/claim`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify({ workspace: workspaceId }),
    });
    if (res.status === 409) {
      const data = (await res.json().catch(() => ({}))) as { assignee?: string | null };
      setNotice(`Already claimed by ${data.assignee || 'another agent'}`);
    }
    await refreshConversations(workspaceId);
  }

      method: 'POST',
      headers: { 'Content-Type': 'application/json' },

  }

  if (loadingMe) {

  }

  if (!me) {
    return (
      <main style={{ padding: 32, fontFamily: 'system-ui', maxWidth: 520, margin: '64px auto' }}>
        <h1>Switchboard</h1>
        <p>Sign in with the Mock IdP to open your workspace inbox.</p>
        <a href="/auth/login">Sign in with IdP</a>
      </main>
    );
  }

  const activeConversation = conversations.find((c) => c.contact === selectedContact) || null;

  return (
    <div style={{ display: 'flex', height: '100vh', fontFamily: 'system-ui' }}>
      <aside style={{ width: 320, borderRight: '1px solid #e5e7eb', padding: 16, overflowY: 'auto' }}>
        <div style={{ marginBottom: 14 }}>
          <div style={{ fontWeight: 600 }}>{me.sub}</div>
          <select
            value={workspaceId}
            onChange={(e) => {
              setSelectedContact('');
              setWorkspaceId(e.target.value);
            }}
            style={{ marginTop: 8, padding: 6 }}
          >
            {me.workspaces.map((w) => (
              <option key={w.id} value={w.id}>
                {w.id} ({w.role})
              </option>
            ))}
          </select>
        </div>

        {notice && (
          <div style={{ color: '#b91c1c', fontSize: 13, marginBottom: 10 }}>{notice}</div>
        )}

        {conversations.map((conv) => (
          <div
            key={conv.contact}
            onClick={() => setSelectedContact(conv.contact)}
            style={{
              padding: 10,
              borderRadius: 6,
              marginBottom: 6,
              cursor: 'pointer',
              background: conv.contact === selectedContact ? '#eef2ff' : '#f9fafb',
            }}
          >
            <div style={{ fontWeight: 600 }}>{conv.contact}</div>
            <div style={{ fontSize: 12, color: '#4b5563' }}>
              {conv.assignee ? `Claimed: ${conv.assignee}` : 'Unassigned'} · {conv.status}
            </div>
          </div>

      <main style={{ flex: 1, display: 'flex', flexDirection: 'column', padding: 16 }}>
        {!selectedContact ? (
          <div style={{ color: '#6b7280' }}>Select a conversation to view messages and reply.</div>
        ) : (
          <>
            <header
              style={{
                display: 'flex',
```

### Fixes to Apply

#### 1. Keep `seq` strictly increasing when published on the WebSocket (`store.ts`)

In `store.ts`, serialize or order event publishing per workspace (e.g., a simple per-workspace promise chain in memory + ordering by `seq` on `publishWorkspaceEvent` so `seq: N` is published before `seq: N+1` starts publishing). Even better: chain `publishWorkspaceEvent` per `workspaceId` so `incr` + Mongo insert + Redis `publish` happen in strict `seq` order on that instance, and have `attachWebSocket` / client ignore out-of-order frames or flush in `seq` order.

Replace `publishWorkspaceEvent` in `scaffold/server/src/store.ts` with:

```ts
const publishChains = new Map<string, Promise<unknown>>();

export function publishWorkspaceEvent(
  workspaceId: string,
  type: EventDoc['type'],
  payload: Record<string, unknown>
): Promise<{ seq: number; frame: Record<string, unknown> }> {
  const prev = publishChains.get(workspaceId) ?? Promise.resolve();
  const next = prev
    .catch(() => {})
    .then(async () => {
      await ensureIndexes();
      const db = await getDb();
      const redis = getRedis();

      const seq = await redis.incr(`switchboard:seq:${workspaceId}`);
      const frame = { seq, type, ...payload };

      await db.collection<EventDoc>('events').insertOne({
        workspaceId,
        seq,
        type,
        payload,
      });

      await redis.publish(wsChannel(workspaceId), JSON.stringify(frame));
      return { seq, frame };
    });

  publishChains.set(workspaceId, next);
  return next;
}

```

#### 2. Prevent unhandled errors from crashing instances & inspect `.check-logs` (`index.ts`)

Add `process.on('unhandledRejection')` and `process.on('uncaughtException')` logging in `scaffold/server/src/index.ts` before `start()`, and ensure Mongo/Redis connections don't drop silently if a connection blips during chaos/crash tests.

```ts
process.on('unhandledRejection', (err) => {
  console.error('unhandledRejection:', err);
});

process.on('uncaughtException', (err) => {
  console.error('uncaughtException:', err);
});

```

Also, in `db.ts`, keep a single shared `MongoClient` promise so concurrent `getDb()` calls during startup never race on `client.connect()`:

```ts
import { MongoClient, type Db } from 'mongodb';
import { config } from './config.js';

let clientPromise: Promise<MongoClient> | undefined;

export async function getDb(): Promise<Db> {
  if (!clientPromise) {
    const client = new MongoClient(config.mongoUrl);
    clientPromise = client.connect();
  }
  const connected = await clientPromise;
  return connected.db();
}

```

That log makes the cause of the `502` crystal clear: `Error: listen EADDRINUSE: address already in use :::4001` (and `:::4002`).

On Windows/MINGW64, `check.js` spawns `sh -c "npm --prefix scaffold/server start"` with `detached: true` and calls `process.kill(-inst.child.pid, 'SIGKILL')`. Under MSYS2/MINGW64, killing the negative PID doesn't always terminate the grandchild `tsx` / `node` process holding port `4001` or `4002`—so when the next check or restart runs, the old `tsx`/`node` process is still bound to `4001`/`4002`.

### How to Fix It

1. **Kill any lingering Node/tsx processes holding ports 4001–4005** before running the check:
In Git Bash / MINGW64:
```bash
for p in 4001 4002 4003 4004 4005; do
  pid=$(netstat -ano 2>/dev/null | awk -v port=":$p" '$2 ~ port && $4 == "LISTENING" {print $5; exit}')
  if [ -n "$pid" ] && [ "$pid" != "0" ]; then
    taskkill //F //PID "$pid" 2>/dev/null || true
  fi
done

```


*(Or in PowerShell: `Get-NetTCPConnection -LocalPort 4001,4002 -ErrorAction SilentlyContinue | ForEach-Object { Stop-Process -Id $_.OwningProcess -Force }`.)*
2. **Start the built server directly in `switchboard.json` instead of `npm` -> `tsx`:**
`npm` spawns a wrapper shell and `tsx` spawns a child `node` process, creating an extra process layer on Windows. Building once and running `node` directly makes process teardown much cleaner:


```json
{
  "install": "npm run scaffold:install && npm --prefix scaffold/server run build",
  "start": "node scaffold/server/dist/index.js",
  "health": "/healthz"
}

```
